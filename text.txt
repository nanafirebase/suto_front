/**
 * INVENTORY RULE
 *
 * productInStock.baseQuantity
 *     =
 * SUM(stockInItem.remainingQuantity)
 *
 * for the same:
 *
 * businessID + branchID + stockID + productID
 *
 * Every operation that changes inventory must happen inside
 * ONE database transaction.
 */


/* ============================================================
 * Helpers
 * ============================================================ */

function toPositiveNumber(value: unknown, field: string): number {
    const number = Number(value);

    if (!Number.isFinite(number) || number <= 0) {
        throw new Error(`Invalid ${field}`);
    }

    return number;
}

function toNonNegativeNumber(value: unknown, field: string): number {
    const number = Number(value);

    if (!Number.isFinite(number) || number < 0) {
        throw new Error(`Invalid ${field}`);
    }

    return number;
}


/* ============================================================
 * RECORD INVENTORY MOVEMENT
 *
 * IMPORTANT:
 * This function assumes the caller already owns the transaction.
 *
 * Do NOT begin/commit a transaction here if consumeStock()
 * is calling this function.
 * ============================================================ */

export async function recordInventoryMovement(
    connection: PoolConnection,
    schema: string,
    movement: {
        id: number;
        businessID: number;
        branchID: number;
        stockID: number;
        productID: number;
        created_by: number;
        referenceType: referenceType;
        referenceID?: number;
        quantityIn?: number;
        quantityOut?: number;
        baseQuantity?: number;
        costPrice?: number;
        currencySnapshotID?: number;
        sessionID?: number | string | null;
        createdAt?: string;
    }
) {
    const movementTable =
        new InventoryMovementTable(undefined, connection, schema);

    const productInStockTable =
        new ProductInStockTable(undefined, connection, schema);

    const quantityIn =
        movement.quantityIn === undefined
            ? 0
            : Number(movement.quantityIn);

    const quantityOut =
        movement.quantityOut === undefined
            ? 0
            : Number(movement.quantityOut);

    if (!Number.isFinite(quantityIn) || quantityIn < 0) {
        throw new Error("Invalid quantityIn");
    }

    if (!Number.isFinite(quantityOut) || quantityOut < 0) {
        throw new Error("Invalid quantityOut");
    }

    if (quantityIn === 0 && quantityOut === 0) {
        throw new Error(
            "Inventory movement must have a quantity"
        );
    }

    if (quantityIn > 0 && quantityOut > 0) {
        throw new Error(
            "Inventory movement cannot have both quantityIn and quantityOut"
        );
    }

    /*
     * NEVER trust baseQuantity supplied by the caller.
     *
     * Calculate it from quantityIn / quantityOut.
     */
    const baseQuantity =
        quantityIn - quantityOut;

    if (!Number.isFinite(baseQuantity) || baseQuantity === 0) {
        throw new Error("Invalid base quantity");
    }


    /* ========================================================
     * Lock aggregate stock row
     * ======================================================== */

    const [stockRows]: any =
        await connection.query(
            `
            SELECT *
            FROM \`${schema}\`.\`productInStock\`
            WHERE businessID = ?
              AND branchID = ?
              AND stockID = ?
              AND productID = ?
            LIMIT 1
            FOR UPDATE
            `,
            [
                movement.businessID,
                movement.branchID,
                movement.stockID,
                movement.productID
            ]
        );


    const currentStock =
        stockRows.length
            ? Number(stockRows[0].baseQuantity || 0)
            : 0;

    if (
        !Number.isFinite(currentStock) ||
        currentStock < 0
    ) {
        throw new Error(
            `Invalid current stock for product ${movement.productID}`
        );
    }

    const newStock =
        currentStock + baseQuantity;


    /* ========================================================
     * Prevent negative inventory
     * ======================================================== */

    if (newStock < 0) {
        throw new Error(
            `Stock cannot go negative. ` +
            `Available: ${currentStock}, ` +
            `requested: ${Math.abs(baseQuantity)}`
        );
    }


    /* ========================================================
     * Save immutable inventory movement
     * ======================================================== */

    movementTable.setValues({
        ...movement,

        quantityIn,
        quantityOut,

        /*
         * Always use calculated value.
         */
        baseQuantity,

        createdAt:
            movement.createdAt || fullDateTime()
    });

    const movementResult =
        await movementTable.save();

    if (movementResult.type !== "success") {
        throw new Error(
            "Failed saving inventory movement"
        );
    }


    /* ========================================================
     * Update / create aggregate stock
     * ======================================================== */

    if (stockRows.length) {

        productInStockTable.setValues({
            id: stockRows[0].id,
            baseQuantity: newStock,
            updatedAt: fullDateTime()
        });

        const updateResult =
            await productInStockTable.update([
                "baseQuantity",
                "updatedAt"
            ]);

        if (updateResult !== "success") {
            throw new Error(
                "Failed updating product stock"
            );
        }

    } else {

        /*
         * We should never be able to create a new aggregate
         * stock record from a negative movement.
         */
        if (newStock < 0) {
            throw new Error(
                "Cannot create stock with negative balance"
            );
        }

        productInStockTable.setValues({
            id: generateID(),

            businessID:
                movement.businessID,

            branchID:
                movement.branchID,

            stockID:
                movement.stockID,

            productID:
                movement.productID,

            baseQuantity:
                newStock,

            reorderLevel:
                15,

            status:
                "active",

            sessionID:
                movement.sessionID,

            createdAt:
                fullDateTime(),

            updatedAt:
                fullDateTime()
        });

        const saveResult =
            await productInStockTable.save();

        if (saveResult.type !== "success") {
            throw new Error(
                "Failed creating product stock"
            );
        }
    }


    return {
        success: true,

        movementID:
            movement.id,

        previousStock:
            currentStock,

        movementQuantity:
            baseQuantity,

        currentStock:
            newStock
    };
}


/* ============================================================
 * CONSUME STOCK
 *
 * FIFO / FEFO
 *
 * EVERYTHING happens inside ONE transaction.
 * ============================================================ */

export async function consumeStock(
    connection: PoolConnection,
    schema: string,
    data: {
        businessID: number;
        branchID: number;
        stockID: number;

        saleID: number;
        salesItemID: number;

        productID: number;
        quantity: number;

        created_by: number;

        strategy?: "FIFO" | "FEFO";

        currencySnapshotID?: number;

        sessionID?: number | string | null;
    }
) {
    const stockInItemTable =
        new StockInItemTable(
            undefined,
            connection,
            schema
        );

    const saleBatchConsumptionTable =
        new SaleBatchConsumptionTable(
            undefined,
            connection,
            schema
        );


    const quantityRequired =
        toPositiveNumber(
            data.quantity,
            "quantity"
        );

    const productID =
        Number(data.productID);

    if (
        !Number.isInteger(productID) ||
        productID <= 0
    ) {
        throw new Error(
            "Invalid product ID"
        );
    }

    if (
        !Number.isInteger(data.saleID) ||
        data.saleID <= 0
    ) {
        throw new Error(
            "Invalid sale ID"
        );
    }

    if (
        !Number.isInteger(data.salesItemID) ||
        data.salesItemID <= 0
    ) {
        throw new Error(
            "Invalid sales item ID"
        );
    }


    /* ========================================================
     * BEGIN TRANSACTION
     * ======================================================== */

    await connection.beginTransaction();

    try {

        /* ====================================================
         * 1. LOCK PRODUCT STOCK
         * ==================================================== */

        const [stockRows]: any =
            await connection.query(
                `
                SELECT *
                FROM \`${schema}\`.\`productInStock\`
                WHERE businessID = ?
                  AND branchID = ?
                  AND stockID = ?
                  AND productID = ?
                LIMIT 1
                FOR UPDATE
                `,
                [
                    data.businessID,
                    data.branchID,
                    data.stockID,
                    productID
                ]
            );


        if (!stockRows.length) {
            throw new Error(
                "Product is out of stock"
            );
        }


        const stock =
            stockRows[0];

        const currentStock =
            Number(
                stock.baseQuantity || 0
            );


        if (
            !Number.isFinite(currentStock) ||
            currentStock < 0
        ) {
            throw new Error(
                "Invalid product stock balance"
            );
        }


        if (
            currentStock < quantityRequired
        ) {
            throw new Error(
                `Insufficient stock. ` +
                `Available: ${currentStock}, ` +
                `requested: ${quantityRequired}`
            );
        }


        /* ====================================================
         * 2. GET + LOCK AVAILABLE BATCHES
         *
         * IMPORTANT:
         * FOR UPDATE prevents another sale from consuming
         * the same batch simultaneously.
         * ==================================================== */

        const [batches]: any =
            await connection.query(
                `
                SELECT
                    sii.*,
                    si.branchID,
                    si.stockID

                FROM \`${schema}\`.\`stockInItem\` sii

                INNER JOIN \`${schema}\`.\`stockIn\` si
                    ON si.id = sii.stockInID

                WHERE sii.businessID = ?
                  AND si.branchID = ?
                  AND si.stockID = ?
                  AND sii.productID = ?
                  AND sii.status = 'active'
                  AND sii.remainingQuantity > 0

                ORDER BY
                    sii.expiryDate ASC,
                    sii.createdAt ASC

                FOR UPDATE
                `,
                [
                    data.businessID,
                    data.branchID,
                    data.stockID,
                    productID
                ]
            );


        if (!batches.length) {
            throw new Error(
                "Product has no available stock batches"
            );
        }


        /* ====================================================
         * 3. CALCULATE TOTAL AVAILABLE BATCH STOCK
         * ==================================================== */

        let batchAvailable = 0;

        for (const batch of batches) {

            const remaining =
                Number(
                    batch.remainingQuantity || 0
                );

            if (
                !Number.isFinite(remaining) ||
                remaining < 0
            ) {
                throw new Error(
                    `Invalid remaining quantity ` +
                    `for stock batch ${batch.id}`
                );
            }

            batchAvailable += remaining;
        }


        /*
         * This is an important invariant.
         *
         * Aggregate stock MUST equal batch stock.
         */
        if (
            batchAvailable !== currentStock
        ) {
            throw new Error(
                `Inventory inconsistency for product ${productID}. ` +
                `ProductInStock: ${currentStock}, ` +
                `available batches: ${batchAvailable}`
            );
        }


        if (
            batchAvailable < quantityRequired
        ) {
            throw new Error(
                `Insufficient batch stock. ` +
                `Available: ${batchAvailable}, ` +
                `requested: ${quantityRequired}`
            );
        }


        /* ====================================================
         * 4. SORT BATCHES
         * ==================================================== */

        const strategy =
            data.strategy || "FEFO";


        if (strategy === "FEFO") {

            batches.sort(
                (a: any, b: any) => {

                    const aExpiry =
                        a.expiryDate
                            ? new Date(
                                a.expiryDate
                            ).getTime()
                            : Number.POSITIVE_INFINITY;

                    const bExpiry =
                        b.expiryDate
                            ? new Date(
                                b.expiryDate
                            ).getTime()
                            : Number.POSITIVE_INFINITY;


                    if (
                        aExpiry !== bExpiry
                    ) {
                        return (
                            aExpiry -
                            bExpiry
                        );
                    }


                    return (
                        new Date(
                            a.createdAt
                        ).getTime() -
                        new Date(
                            b.createdAt
                        ).getTime()
                    );
                }
            );

        } else {

            batches.sort(
                (a: any, b: any) =>
                    new Date(
                        a.createdAt
                    ).getTime() -
                    new Date(
                        b.createdAt
                    ).getTime()
            );
        }


        /* ====================================================
         * 5. CONSUME BATCHES
         * ==================================================== */

        let quantityRemaining =
            quantityRequired;

        let totalCost = 0;


        const consumptions: {
            stockInID: number;
            stockInItemID: number;
            quantity: number;
            costPrice: number;
            totalCost: number;
        }[] = [];


        for (
            const batch of batches
        ) {

            if (
                quantityRemaining <= 0
            ) {
                break;
            }


            const stockInItemID =
                Number(batch.id);


            const stockInID =
                Number(batch.stockInID);


            const available =
                Number(
                    batch.remainingQuantity || 0
                );


            if (
                !Number.isInteger(
                    stockInItemID
                ) ||
                stockInItemID <= 0
            ) {
                throw new Error(
                    "Invalid stock-in item ID"
                );
            }


            if (
                !Number.isInteger(
                    stockInID
                ) ||
                stockInID <= 0
            ) {
                throw new Error(
                    "Invalid stock-in ID"
                );
            }


            if (
                !Number.isFinite(
                    available
                ) ||
                available <= 0
            ) {
                continue;
            }


            const quantityConsumed =
                Math.min(
                    quantityRemaining,
                    available
                );


            const costPrice =
                Number(
                    batch.costPrice || 0
                );


            if (
                !Number.isFinite(
                    costPrice
                ) ||
                costPrice < 0
            ) {
                throw new Error(
                    `Invalid cost price ` +
                    `for stock batch ${stockInItemID}`
                );
            }


            const batchCost =
                quantityConsumed *
                costPrice;


            const remainingQuantity =
                available -
                quantityConsumed;


            /* ================================================
             * Update batch
             * ================================================ */

            const [updateResult]: any =
                await connection.query(
                    `
                    UPDATE \`${schema}\`.\`stockInItem\`
                    SET remainingQuantity = ?
                    WHERE id = ?
                      AND remainingQuantity >= ?
                    `,
                    [
                        remainingQuantity,
                        stockInItemID,
                        quantityConsumed
                    ]
                );


            if (
                updateResult.affectedRows !== 1
            ) {
                throw new Error(
                    `Failed updating stock batch ${stockInItemID}`
                );
            }


            /* ================================================
             * Record batch consumption
             * ================================================ */

            saleBatchConsumptionTable.setValues({

                id:
                    generateID(),

                businessID:
                    data.businessID,

                branchID:
                    data.branchID,

                saleID:
                    data.saleID,

                saleItemID:
                    data.salesItemID,

                stockInID:
                    stockInID,

                stockInItemID:
                    stockInItemID,

                productID:
                    productID,

                quantity:
                    quantityConsumed,

                costPrice:
                    costPrice,

                totalCost:
                    batchCost,

                status:
                    "active",

                sessionID:
                    data.sessionID,

                createdAt:
                    fullDateTime()
            });


            const saveResult =
                await saleBatchConsumptionTable.save();


            if (
                saveResult.type !== "success"
            ) {
                throw new Error(
                    `Failed recording batch consumption ` +
                    `for stock batch ${stockInItemID}`
                );
            }


            consumptions.push({

                stockInID,

                stockInItemID,

                quantity:
                    quantityConsumed,

                costPrice,

                totalCost:
                    batchCost
            });


            totalCost +=
                batchCost;

            quantityRemaining -=
                quantityConsumed;
        }


        /* ====================================================
         * 6. FINAL SAFETY CHECK
         * ==================================================== */

        if (
            quantityRemaining > 0
        ) {
            throw new Error(
                `Unable to consume complete quantity. ` +
                `Remaining: ${quantityRemaining}`
            );
        }


        /* ====================================================
         * 7. UPDATE AGGREGATE + LEDGER
         *
         * recordInventoryMovement() uses the SAME
         * transaction because it receives this connection.
         * ==================================================== */

        const movementResult =
            await recordInventoryMovement(
                connection,
                schema,
                {
                    id:
                        generateID(),

                    businessID:
                        data.businessID,

                    branchID:
                        data.branchID,

                    stockID:
                        data.stockID,

                    productID:
                        productID,

                    created_by:
                        data.created_by,

                    referenceType:
                        "sale",

                    referenceID:
                        data.saleID,

                    quantityIn:
                        0,

                    quantityOut:
                        quantityRequired,

                    /*
                     * Do not rely on this value inside
                     * recordInventoryMovement().
                     *
                     * It calculates:
                     * 0 - quantityRequired
                     */
                    baseQuantity:
                        -quantityRequired,

                    currencySnapshotID:
                        data.currencySnapshotID,

                    sessionID:
                        data.sessionID,

                    createdAt:
                        fullDateTime()
                }
            );


        if (
            !movementResult.success
        ) {
            throw new Error(
                "Failed recording inventory movement"
            );
        }


        /* ====================================================
         * 8. VERIFY FINAL BALANCE
         *
         * Recalculate batch balance while transaction is
         * still open.
         * ==================================================== */

        const [finalBatchRows]: any =
            await connection.query(
                `
                SELECT
                    COALESCE(
                        SUM(sii.remainingQuantity),
                        0
                    ) AS totalAvailable

                FROM \`${schema}\`.\`stockInItem\` sii

                INNER JOIN \`${schema}\`.\`stockIn\` si
                    ON si.id = sii.stockInID

                WHERE sii.businessID = ?
                  AND si.branchID = ?
                  AND si.stockID = ?
                  AND sii.productID = ?
                  AND sii.status = 'active'
                `,
                [
                    data.businessID,
                    data.branchID,
                    data.stockID,
                    productID
                ]
            );


        const finalBatchStock =
            Number(
                finalBatchRows[0]?.totalAvailable || 0
            );


        const expectedStock =
            currentStock -
            quantityRequired;


        if (
            finalBatchStock !==
            expectedStock
        ) {
            throw new Error(
                `Final inventory mismatch for product ${productID}. ` +
                `Expected: ${expectedStock}, ` +
                `batches: ${finalBatchStock}`
            );
        }


        /* ====================================================
         * 9. COMMIT
         * ==================================================== */

        await connection.commit();


        return {
            success: true,

            productID,

            requested:
                quantityRequired,

            consumed:
                quantityRequired,

            previousStock:
                currentStock,

            remainingStock:
                expectedStock,

            totalCost,

            consumptions
        };


    } catch (error) {

        /*
         * ANY failure rolls back:
         *
         * - batch changes
         * - saleBatchConsumption
         * - inventoryMovement
         * - productInStock
         */
        await connection.rollback();

        throw error;
    }
}

                    .filter((i: any) => i.taxGroupID === group.id)
                    .sort((a: any, b: any) => a.order - b.order)

                map[group.id] = items.map((item: any) => {
                    const tax = taxes.find((t: any) => t.id === item.taxID)

                    return {
                        id: tax.id,
                        taxName: tax.taxName,
                        percentage: tax.percentage,
                        calculationType: tax.calculationType,
                        order: item.order
                    }
                })
            })

            setTaxMap(map)
        }
    })
}

const handleAddItem = (product: any) => {
    if (!product) return

    const newItem: CartItem = {
        productID: product.id,
        name: product.productName,
        price: product.price,
        qty: 1,
        total: product.price,
        taxGroupID: Number(product.taxGroupID) || null
    }

    setCart(prev => [...prev, newItem])
}

// ---------------- UPDATE QTY ----------------

const updateQty = (index: number, qty: number) => {
    const updated = [...cart]

    updated[index].qty = qty
    updated[index].total = qty * updated[index].price

    setCart(updated)
}

// ---------------- TAX CALCULATION ----------------

const taxSummary = useMemo(() => {
    let totalTax = 0
    let breakdown: any[] = []

    cart.forEach(item => {
        if (!item.taxGroupID) return

        const taxes = taxMap[item.taxGroupID] || []
        let baseAmount = item.total

        taxes.forEach((tax) => {
            let taxAmount = 0

            if (tax.calculationType === 'compound') {
                taxAmount = (baseAmount * tax.percentage) / 100
                baseAmount += taxAmount
            } else {
                taxAmount = (item.total * tax.percentage) / 100
            }

            totalTax += taxAmount

            const existing = breakdown.find(t => t.id === tax.id)

            if (existing) {
                existing.amount += taxAmount
            } else {
                breakdown.push({
                    id: tax.id,
                    name: tax.taxName,
                    percentage: tax.percentage,
                    amount: taxAmount
                })
            }
        })
    })

    return { breakdown, totalTax }
}, [cart, taxMap])

// ---------------- TOTAL ----------------

const subtotal = useMemo(() => {
    return cart.reduce((sum, item) => sum + item.total, 0)
}, [cart])

const grandTotal = useMemo(() => {
    return subtotal + taxSummary.totalTax
}, [subtotal, taxSummary])

    }

    }

    SocketIO.emit('save-sale', payload, (response: any) => {
        if (response.status === "success") {
            Alert.alert("Success", response.message)
            setCart([])
        } else {
            Alert.alert("Error", response.message || "Failed")
        }
    })
}

import Svg, { Circle } from 'react-native-svg';

type ChartItem = {
    label: string;
    value: number;
    color: string;
};

type DonutChartProps = {
    data: ChartItem[];
    total?: number;
    size?: number;
    strokeWidth?: number;
};

const DonutChart = ({
    data,
    total,
    size = 220,
    strokeWidth = 32,
}: DonutChartProps) => {
    const center = size / 2;
    const radius = (size - strokeWidth) / 2;
    const circumference = 2 * Math.PI * radius;

    const chartTotal = total ?? data.reduce((sum, item) => sum + item.value, 0);

    // Space between segments
    const gap = 8;

    let currentOffset = 0;

    return (
        <View
            style={{
                width: size,
                height: size,
                alignItems: 'center',
                justifyContent: 'center',
            }}
        >
            <Svg width={size} height={size}>
                {data.map((item, index) => {
                    const percentage = item.value / chartTotal;

                    const segmentLength =
                        circumference * percentage - gap;

                    const dashArray = `${segmentLength} ${
                        circumference - segmentLength
                    }`;

                    const rotation =
                        (currentOffset / circumference) * 360 - 90;

                    currentOffset += circumference * percentage;

                    return (
                        <Circle
                            key={index}
                            cx={center}
                            cy={center}
                            r={radius}
                            stroke={item.color}
                            strokeWidth={strokeWidth}
                            fill="none"
                            strokeDasharray={dashArray}
                            strokeDashoffset={0}
                            strokeLinecap="round"
                            rotation={rotation}
                            origin={`${center}, ${center}`}
                        />
                    );
                })}
            </Svg>

            {/* Center content */}
            <View
                style={{
                    position: 'absolute',
                    alignItems: 'center',
                }}
            >
                <Text
                    style={{
                        color: '#777',
                        fontSize: 14,
                    }}
                >
                    Take home
                </Text>

                <Text
                    style={{
                        color: '#111',
                        fontSize: 20,
                        fontWeight: '600',
                        marginTop: 4,
                    }}
                >
                    $1,232.03
                </Text>
            </View>
        </View>
    );
};

export default DonutChart;
